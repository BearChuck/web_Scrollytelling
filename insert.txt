  (function () {
    // 動態載入外部相依套件 (若宿主網站尚未載入 Tailwind 或 GSAP)
    function loadScript(src) {
      return new Promise(function (resolve, reject) {
        var existing = document.querySelector('script[src="' + src + '"]');
        if (existing) {
          resolve();
          return;
        }
        var s = document.createElement('script');
        s.src = src;
        s.onload = resolve;
        s.onerror = reject;
        document.head.appendChild(s);
      });
    }

    async function ensureDependenciesAndInit() {
      if (!window.tailwind) {
        await loadScript('https://cdn.tailwindcss.com');
      }
      if (!window.gsap) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/gsap.min.js');
      }
      if (!window.ScrollTrigger) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollTrigger.min.js');
      }
      if (!window.ScrollToPlugin) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollToPlugin.min.js');
      }
      initEventOpticsWidget();
    }

    function initEventOpticsWidget() {
      var root = document.getElementById('eo-scrolly-widget');
      if (!root) return;

      gsap.registerPlugin(ScrollTrigger, ScrollToPlugin);

      var phasesData = [
        {

          bgScale: 1.15,
          bgBlur: 1.5,
          bokehOpacity: 0.32,
          coneSpread: 30
        },
        {
          index: 2,
          focal: 85,
          fov: 28.5,
          dist: '2.8m',
          compression: 74,
          lensLabel: '85mm f/1.4 GM II',
          compressionTag: '雜物剝離: 舞台散景',
          color: '#a855f7',
          glowClass: 'bg-purple-500/10',
          bgScale: 1.75,
          bgBlur: 4.2,
          bokehOpacity: 0.68,
          coneSpread: 16
        },
        {
          index: 3,
          focal: 200,
          fov: 12,
          dist: '6.5m',
          compression: 98,
          lensLabel: '200mm f/2.8 GM',
          compressionTag: '背板壓縮: 滿版貼合',
          color: '#fbbf24',
          glowClass: 'bg-amber-500/10',
          bgScale: 2.65,
          bgBlur: 8.0,

          coneSpread: 7
        }
      ];

      var activeTrack = root.querySelector('#eo-svg-active-track');
      var playhead = root.querySelector('#eo-svg-playhead');
      var playheadCone = root.querySelector('#eo-playhead-fov-cone');

      var topProgressText = root.querySelector('#eo-top-progress-text');
      var headerFocalReadout = root.querySelector('#eo-header-focal-readout');

      var vfLensLabel = root.querySelector('#eo-vf-lens-label');
      var vfDist = root.querySelector('#eo-vf-dist');
      var vfBgLayer = root.querySelector('#eo-vf-bg-layer');
      var vfBokehGroup = root.querySelector('#eo-vf-bokeh-group');
      var vfCompressionTag = root.querySelector('#eo-vf-compression-tag');
      var ambientGlow = root.querySelector('#eo-ambient-glow');

      var metricFocal = root.querySelector('#eo-metric-focal');
      var metricFov = root.querySelector('#eo-metric-fov');
      var metricCompression = root.querySelector('#eo-metric-compression');

      var svgNodes = root.querySelectorAll('.eo-timeline-node');
      var phaseCards = root.querySelectorAll('[data-eo-card]');
      var hudButtons = root.querySelectorAll('[data-eo-hud]');

      var pathLength = activeTrack.getTotalLength();
      activeTrack.style.strokeDasharray = String(pathLength);
      activeTrack.style.strokeDashoffset = String(pathLength);

      var nodeTargets = [
        { x: 200, y: 55 },
        { x: 160, y: 230 },
        { x: 240, y: 380 },
        { x: 200, y: 545 }
      ];

      var nodeRatios = nodeTargets.map(function (target) {
        var bestLength = 0;
        var minDistance = Infinity;
        var steps = 400;
        for (var i = 0; i <= steps; i++) {
          var len = (i / steps) * pathLength;
          var pt = activeTrack.getPointAtLength(len);
          var dist = Math.hypot(pt.x - target.x, pt.y - target.y);
          if (dist < minDistance) {
            minDistance = dist;
            bestLength = len;
          }
        }
        return bestLength / pathLength;
      });

      nodeRatios[0] = 0;
      nodeRatios[3] = 1;

      var currentActiveIndex = -1;

      function interpolateMetrics(progress) {
        var clamped = Math.max(0, Math.min(1, progress));
        var segIndex = 0;
        for (var i = 0; i < nodeRatios.length - 1; i++) {
          if (clamped >= nodeRatios[i] && clamped <= nodeRatios[i + 1]) {
            segIndex = i;
            break;
          }
        }
        var startRatio = nodeRatios[segIndex];
        var endRatio = nodeRatios[segIndex + 1];
        var localT = endRatio > startRatio ? (clamped - startRatio) / (endRatio - startRatio) : 0;

        var pA = phasesData[segIndex];
        var pB = phasesData[segIndex + 1];
        var lerp = function (a, b, t) { return a + (b - a) * t; };

        return {
          focal: Math.round(lerp(pA.focal, pB.focal, localT)),
          fov: lerp(pA.fov, pB.fov, localT).toFixed(1),
          compression: Math.round(lerp(pA.compression, pB.compression, localT)),
          bgScale: lerp(pA.bgScale, pB.bgScale, localT),
          bgBlur: lerp(pA.bgBlur, pB.bgBlur, localT),
          bokehOpacity: lerp(pA.bokehOpacity, pB.bokehOpacity, localT),
          coneSpread: lerp(pA.coneSpread, pB.coneSpread, localT)
        };
      }

      function setActivePhase(index) {
        if (currentActiveIndex === index) return;
        currentActiveIndex = index;
        var activeData = phasesData[index];

        vfLensLabel.textContent = activeData.lensLabel;
        if (vfDist) vfDist.textContent = activeData.dist;
        vfCompressionTag.textContent = activeData.compressionTag;

        ambientGlow.className = 'absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[340px] sm:w-[560px] h-[340px] sm:h-[560px] rounded-full blur-[130px] pointer-events-none transition-colors duration-700 ' + activeData.glowClass;

        svgNodes.forEach(function (node, i) {
          var outer = node.querySelector('.node-outer');
          var core = node.querySelector('.node-core');
          if (i <= index) {
            node.classList.add('is-active');
            outer.setAttribute('stroke', phasesData[i].color);
            core.setAttribute('fill', phasesData[i].color);
          } else {
            node.classList.remove('is-active');
            outer.setAttribute('stroke', '#475569');
            core.setAttribute('fill', '#94a3b8');
          }
        });

        phaseCards.forEach(function (card) {
          var cardIdx = parseInt(card.getAttribute('data-eo-card'), 10);
          if (cardIdx === index) {
            card.classList.add('is-active');
            gsap.to(card, {
              opacity: 1,
              scale: 1,
              y: 0,
              duration: 0.38,
              ease: 'power3.out',
              overwrite: 'auto'
            });
          } else {
            card.classList.remove('is-active');
            gsap.to(card, {
              opacity: 0,
              scale: 0.97,
              y: 12,
              duration: 0.3,
              ease: 'power2.out',
              overwrite: 'auto'
            });
          }
        });

        hudButtons.forEach(function (btn, i) {
          if (i === index) {
            btn.classList.add('is-active');
            btn.classList.remove('text-slate-400');
          } else {
            btn.classList.remove('is-active');
            btn.classList.add('text-slate-400');
          }
        });
      }

      function updateStageOnScroll(progress) {
        var clamped = Math.max(0, Math.min(1, progress));

        topProgressBar.style.transform = 'scaleX(' + clamped + ')';
        var pct = Math.round(clamped * 100);
        topProgressText.textContent = String(pct).padStart(3, '0') + '%';

        var drawLength = pathLength * clamped;
        activeTrack.style.strokeDashoffset = String(pathLength - drawLength);

        var currentPt = activeTrack.getPointAtLength(drawLength);
        playhead.setAttribute('transform', 'translate(' + currentPt.x.toFixed(1) + ', ' + currentPt.y.toFixed(1) + ')');

        var interp = interpolateMetrics(clamped);
        headerFocalReadout.textContent = interp.focal + 'mm';
        metricFocal.textContent = interp.focal + 'mm';
        metricFov.textContent = interp.fov + '°';
        metricCompression.textContent = interp.compression + '%';

        vfBgLayer.style.transform = 'scale(' + interp.bgScale.toFixed(3) + ')';
        vfBgLayer.style.filter = 'blur(' + interp.bgBlur.toFixed(2) + 'px)';
        vfBokehGroup.setAttribute('opacity', interp.bokehOpacity.toFixed(2));

        var spread = interp.coneSpread.toFixed(1);
        playheadCone.setAttribute('points', '0,0 -' + spread + ',-48 ' + spread + ',-48');

        var closestIndex = 0;
        var minDiff = Infinity;
        nodeRatios.forEach(function (ratio, idx) {
          var diff = Math.abs(clamped - ratio);
          if (diff < minDiff) {
            minDiff = diff;
            closestIndex = idx;
          }
        });
        setActivePhase(closestIndex);
      }

      var isMobileViewport = window.matchMedia('(max-width: 1023px)').matches;
      var mainScrollTrigger = ScrollTrigger.create({
        trigger: '#eo-pinned-stage',
        pin: true,
        start: 'top top',
        end: isMobileViewport ? '+=2000' : '+=3000',
        scrub: 1.0,
        onUpdate: function (self) {
          updateStageOnScroll(self.progress);
        }
      });

      setActivePhase(0);
      updateStageOnScroll(0);

      function scrollToPhase(phaseIndex) {
        var targetRatio = nodeRatios[phaseIndex] !== undefined ? nodeRatios[phaseIndex] : 0;
        var targetScrollY = mainScrollTrigger.start + (mainScrollTrigger.end - mainScrollTrigger.start) * targetRatio;

        gsap.to(window, {
          scrollTo: { y: targetScrollY, autoKill: false },
          duration: 0.9,
          ease: 'power3.inOut'
        });
      }

      svgNodes.forEach(function (node) {
        var phaseIndex = parseInt(node.getAttribute('data-eo-phase'), 10);
        node.addEventListener('click', function () { scrollToPhase(phaseIndex); });
        node.addEventListener('keydown', function (e) {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            scrollToPhase(phaseIndex);
          }
        });
      });

      hudButtons.forEach(function (btn) {
        var phaseIndex = parseInt(btn.getAttribute('data-eo-hud'), 10);
        btn.addEventListener('click', function () { scrollToPhase(phaseIndex); });
      });

      phaseCards.forEach(function (card) {
        var phaseIndex = parseInt(card.getAttribute('data-eo-card'), 10);
        card.addEventListener('click', function () { scrollToPhase(phaseIndex); });
      });

      window.addEventListener('resize', function () {
        var idx = currentActiveIndex;
        currentActiveIndex = -1;
        setActivePhase(idx >= 0 ? idx : 0);
        ScrollTrigger.refresh();
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', ensureDependenciesAndInit);
    } else {
      ensureDependenciesAndInit();
    }
  })();
