<script>
  (function () {
    function loadScript(src) {
      return new Promise(function (resolve, reject) {
        if (document.querySelector('script[src="' + src + '"]')) { resolve(); return; }
        var s = document.createElement('script');
        s.src = src; s.onload = resolve; s.onerror = reject;
        document.head.appendChild(s);
      });
    }

    async function ensureDeps() {
      if (!window.tailwind) await loadScript('https://cdn.tailwindcss.com');
      if (!window.gsap) await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/gsap.min.js');
      if (!window.ScrollTrigger) await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollTrigger.min.js');
      if (!window.ScrollToPlugin) await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollToPlugin.min.js');
      requestAnimationFrame(function () { requestAnimationFrame(initBrassWidget); });
    }

    function initBrassWidget() {
      var root = document.getElementById('ba-scrolly-widget');
      if (!root) return;
      gsap.registerPlugin(ScrollTrigger, ScrollToPlugin);

      var phases = [
        { zone: 'ZONE 01 // 橫膈膜高壓氣柱', mode: 'DC AIRFLOW', status: '能量轉換: 橫膈膜穩定加壓', p: '3.2 kPa', f: '0 Hz', spl: '45 dB', color: '#22d3ee', part: 'ba-part-breath', valves: [0,0,0] },
        { zone: 'ZONE 02 // 雙唇伯努利振源', mode: 'LIP BUZZING', status: '能量轉換: 氣流切割為脈衝波', p: '5.8 kPa', f: '233 Hz', spl: '68 dB', color: '#f43f5e', part: 'ba-part-breath', valves: [0,0,0] },
        { zone: 'ZONE 03 // 號嘴杯部與喉孔', mode: 'VENTURI BOOST', status: '能量轉換: 亥姆霍茲聚能加速', p: '8.4 kPa', f: '349 Hz', spl: '82 dB', color: '#fbbf24', part: 'ba-part-mouthpiece', valves: [0,0,0] },
        { zone: 'ZONE 04 // 148cm 圓柱管體駐波', mode: 'STANDING WAVE', status: '能量轉換: 管內駐波泛音共鳴', p: '10.6 kPa', f: '466 Hz', spl: '94 dB', color: '#34d399', part: 'ba-part-bore', valves: [0,0,0] },
        { zone: 'ZONE 05 // ピストン活塞閥門變調', mode: 'VALVE MODULATION', status: '能量轉換: 迂迴管延長改變音高', p: '12.2 kPa', f: '587 Hz', spl: '101 dB', color: '#a855f7', part: 'ba-part-valves', valves: [1,1,0] },
        { zone: 'ZONE 06 // 指數型號口聲學綻放', mode: 'BELL RADIATION', status: '能量轉換: 高頻泛音向空間綻放', p: '14.8 kPa', f: '932 Hz', spl: '112 dB', color: '#fbbf24', part: 'ba-part-bell', valves: [0,0,0] }
      ];

      var stage = root.querySelector('#ba-pinned-stage');
      var activeTrack = root.querySelector('#ba-svg-active-track');
      var playhead = root.querySelector('#ba-svg-playhead');
      var topBar = root.querySelector('#ba-top-progress-bar');
      var topText = root.querySelector('#ba-top-progress-text');
      var headerFreq = root.querySelector('#ba-header-freq');
      var zoneLabel = root.querySelector('#ba-zone-label');
      var waveMode = root.querySelector('#ba-wave-mode');
      var statusTag = root.querySelector('#ba-status-tag');
      var wavePath = root.querySelector('#ba-standing-wave');
      var bellRings = root.querySelector('#ba-bell-rings');
      var mPressure = root.querySelector('#ba-metric-pressure');
      var mFreq = root.querySelector('#ba-metric-freq');
      var mSpl = root.querySelector('#ba-metric-spl');
      var v1 = root.querySelector('#ba-v1'), v2 = root.querySelector('#ba-v2'), v3 = root.querySelector('#ba-v3');
      var nodes = root.querySelectorAll('.ba-timeline-node');
      var cards = root.querySelectorAll('[data-ba-card]');
      var huds = root.querySelectorAll('[data-ba-hud]');

      var pathLength = activeTrack.getTotalLength();
      activeTrack.style.strokeDasharray = String(pathLength);
      activeTrack.style.strokeDashoffset = String(pathLength);

      var targets = [{x:200,y:45},{x:145,y:155},{x:255,y:265},{x:145,y:375},{x:255,y:490},{x:200,y:605}];
      var ratios = targets.map(function (t) {
        var best = 0, minD = Infinity;
        for (var i = 0; i <= 400; i++) {
          var l = (i / 400) * pathLength;
          var p = activeTrack.getPointAtLength(l);
          var d = Math.hypot(p.x - t.x, p.y - t.y);
          if (d < minD) { minD = d; best = l; }
        }
        return best / pathLength;
      });
      ratios[0] = 0; ratios[5] = 1;

      var curIdx = -1;
      function setPhase(idx) {
        if (curIdx === idx) return;
        curIdx = idx;
        var d = phases[idx];
        zoneLabel.textContent = d.zone;
        waveMode.textContent = d.mode;
        statusTag.textContent = d.status;
        headerFreq.textContent = d.f;
        mPressure.textContent = d.p;
        mFreq.textContent = d.f;
        mSpl.textContent = d.spl;
        ['ba-part-breath','ba-part-mouthpiece','ba-part-bore','ba-part-valves','ba-part-bell'].forEach(function (id) {
          var el = root.querySelector('#' + id);
          if (el) el.setAttribute('opacity', id === d.part ? '1' : '0.35');
        });
        if (v1 && v2 && v3) {
          v1.setAttribute('y', d.valves[0] ? '62' : '52');
          v2.setAttribute('y', d.valves[1] ? '66' : '58');
          v3.setAttribute('y', d.valves[2] ? '62' : '52');
        }
        nodes.forEach(function (n, i) {
          var o = n.querySelector('.node-outer'), c = n.querySelector('.node-core');
          if (i <= idx) { n.classList.add('is-active'); o.setAttribute('stroke', phases[i].color); c.setAttribute('fill', phases[i].color); }
          else { n.classList.remove('is-active'); o.setAttribute('stroke', '#475569'); c.setAttribute('fill', '#94a3b8'); }
        });
        cards.forEach(function (c, i) {
          if (i === idx) { c.classList.add('is-active'); gsap.to(c, { opacity: 1, scale: 1, y: 0, duration: 0.35 }); }
          else { c.classList.remove('is-active'); gsap.to(c, { opacity: 0, scale: 0.97, y: 14, duration: 0.25 }); }
        });
        huds.forEach(function (b, i) {
          b.classList.toggle('is-active', i === idx);
        });
      }

      function onScrollUpdate(prog) {
        var c = Math.max(0, Math.min(1, prog));
        topBar.style.transform = 'scaleX(' + c + ')';
        topText.textContent = String(Math.round(c * 100)).padStart(3, '0') + '%';
        var draw = pathLength * c;
        activeTrack.style.strokeDashoffset = String(pathLength - draw);
        var pt = activeTrack.getPointAtLength(draw);
        playhead.setAttribute('transform', 'translate(' + pt.x.toFixed(1) + ',' + pt.y.toFixed(1) + ')');

        var pts = [];
        for (var i = 0; i <= 50; i++) {
          var t = i / 50;
          var x = 65 + 353 * t;
          var y = 90 + Math.sin(t * Math.PI * 2 * (1.5 + c * 4.5) - c * 16) * (3 + c * 20) * Math.sin(t * Math.PI);
          pts.push((i === 0 ? 'M ' : 'L ') + x.toFixed(1) + ' ' + y.toFixed(1));
        }
        wavePath.setAttribute('d', pts.join(' '));
        bellRings.setAttribute('opacity', (0.1 + c * 0.9).toFixed(2));

        var bestI = 0, minD = Infinity;
        ratios.forEach(function (r, idx) { var d = Math.abs(c - r); if (d < minD) { minD = d; bestI = idx; } });
        setPhase(bestI);
      }

      var st = ScrollTrigger.create({
        trigger: stage,
        pin: true,
        pinSpacing: true,
        start: 'top top',
        end: '+=3200',
        scrub: 1.0,
        onUpdate: function (self) { onScrollUpdate(self.progress); }
      });

      setPhase(0); onScrollUpdate(0);

      function jumpTo(idx) {
        var y = st.start + (st.end - st.start) * (ratios[idx] || 0);
        gsap.to(window, { scrollTo: { y: y, autoKill: false }, duration: 0.9, ease: 'power3.inOut' });
      }
      nodes.forEach(function (n, i) { n.addEventListener('click', function () { jumpTo(i); }); });
      huds.forEach(function (b, i) { b.addEventListener('click', function () { jumpTo(i); }); });
      cards.forEach(function (c, i) { c.addEventListener('click', function () { jumpTo(i); }); });
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', ensureDeps);
    else ensureDeps();
  })();
