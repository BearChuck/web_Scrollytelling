    async function ensureDependenciesAndInit() {
      if (!window.tailwind) {
        await loadScript('https://cdn.tailwindcss.com');
      }
      if (!window.gsap) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/gsap.min.js');
      }
      if (!window.ScrollTrigger) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollTrigger.min.js');
      }
      if (!window.ScrollToPlugin) {
        await loadScript('https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollToPlugin.min.js');
      }
      // 等待 Tailwind CDN 樣式注入後再初始化 ScrollTrigger，確保高度計算 100% 精準
      requestAnimationFrame(function () {
        requestAnimationFrame(initBrassWidget);
      });
    }

    function initBrassWidget() {
      var root = document.getElementById('ba-scrolly-widget');
      if (!root) return;

      gsap.registerPlugin(ScrollTrigger, ScrollToPlugin);

      var phases = [
        {
          zone: 'ZONE 01 // 橫膈膜高壓氣柱',
          mode: 'DC AIRFLOW',
          status: '能量轉換: 橫膈膜穩定加壓',
          pVal: 3.2,
          fVal: 0,
          splVal: 45,
          note: '直流氣流',
          color: '#22d3ee',
          glowClass: 'bg-cyan-500/12',
          part: 'ba-part-breath',
          valves: [0, 0, 0]
        },
        {
          zone: 'ZONE 02 // 雙唇伯努利振源',
          mode: 'LIP BUZZING',
          status: '能量轉換: 氣流切割為脈衝波',
          pVal: 5.8,
          fVal: 233,
          splVal: 68,
          note: 'Bb3',
          color: '#f43f5e',
          glowClass: 'bg-rose-500/12',
          part: 'ba-part-breath',
          valves: [0, 0, 0]
        },
        {
          zone: 'ZONE 03 // 號嘴杯部與喉孔',
          mode: 'VENTURI BOOST',
          status: '能量轉換: 亥姆霍茲聚能加速',
          pVal: 8.4,
          fVal: 349,
          splVal: 82,
          note: 'F4',
          color: '#fbbf24',
          glowClass: 'bg-amber-500/12',
          part: 'ba-part-mouthpiece',
          valves: [0, 0, 0]
        },
        {
          zone: 'ZONE 04 // 148cm 圓柱管體駐波',
          mode: 'STANDING WAVE',
          status: '能量轉換: 管內駐波泛音共鳴',
          pVal: 10.6,
          fVal: 466,
          splVal: 94,
          note: 'Bb4',
          color: '#34d399',
          glowClass: 'bg-emerald-500/12',
          part: 'ba-part-bore',
          valves: [0, 0, 0]
        },
        {
          zone: 'ZONE 05 // 活塞閥門管長變調',
          mode: 'VALVE MODULATION',
          status: '能量轉換: 迂迴管延長改變音高',
          pVal: 12.2,
          fVal: 587,
          splVal: 101,
          note: 'D5',
          color: '#a855f7',
          glowClass: 'bg-purple-500/12',
          part: 'ba-part-valves',
          valves: [1, 1, 0]
        },
        {
          zone: 'ZONE 06 // 指數型號口聲學綻放',
          mode: 'BELL RADIATION',
          status: '能量轉換: 高頻泛音向空間綻放',
          pVal: 14.8,
          fVal: 932,
          splVal: 112,
          note: 'Bb5',
          color: '#fbbf24',
          glowClass: 'bg-amber-500/16',
          part: 'ba-part-bell',
          valves: [0, 0, 0]
        }
      ];

      var runway = root.querySelector('#ba-scroll-runway');
      var ambientGlow = root.querySelector('#ba-ambient-glow');
      var activeTrack = root.querySelector('#ba-svg-active-track');
      var playhead = root.querySelector('#ba-svg-playhead');
      var topBar = root.querySelector('#ba-top-progress-bar');
      var topText = root.querySelector('#ba-top-progress-text');
      var headerFreq = root.querySelector('#ba-header-freq');
      var zoneLabel = root.querySelector('#ba-zone-label');
      var waveMode = root.querySelector('#ba-wave-mode');
      var statusTag = root.querySelector('#ba-status-tag');
      var wavePath = root.querySelector('#ba-standing-wave');
      var bellRings = root.querySelector('#ba-bell-rings');
      var mPressure = root.querySelector('#ba-metric-pressure');
      var mFreq = root.querySelector('#ba-metric-freq');
      var mSpl = root.querySelector('#ba-metric-spl');
      var v1 = root.querySelector('#ba-v1'), v2 = root.querySelector('#ba-v2'), v3 = root.querySelector('#ba-v3');
      var nodes = root.querySelectorAll('.ba-timeline-node');
      var cards = root.querySelectorAll('[data-ba-card]');
      var huds = root.querySelectorAll('[data-ba-hud]');

      var pathLength = activeTrack.getTotalLength();
      activeTrack.style.strokeDasharray = String(pathLength);
      activeTrack.style.strokeDashoffset = String(pathLength);

      var targets = [{x:200,y:45},{x:145,y:155},{x:255,y:265},{x:145,y:375},{x:255,y:490},{x:200,y:605}];
      var ratios = targets.map(function (t) {
        var best = 0, minD = Infinity;
        for (var i = 0; i <= 400; i++) {
          var l = (i / 400) * pathLength;
          var p = activeTrack.getPointAtLength(l);
          var d = Math.hypot(p.x - t.x, p.y - t.y);
          if (d < minD) { minD = d; best = l; }
        }
        return best / pathLength;
      });
      ratios[0] = 0; ratios[5] = 1;

      // 毫秒級平滑插值演算法 (Continuous Acoustic Metric Interpolation)
      function interpolateMetrics(progress) {
        var clamped = Math.max(0, Math.min(1, progress));
        var segIndex = 0;
        for (var i = 0; i < ratios.length - 1; i++) {
          if (clamped >= ratios[i] && clamped <= ratios[i + 1]) {
            segIndex = i;
            break;
          }
        }
        var startRatio = ratios[segIndex];
        var endRatio = ratios[segIndex + 1];
        var localT = endRatio > startRatio ? (clamped - startRatio) / (endRatio - startRatio) : 0;
        var pA = phases[segIndex];
        var pB = phases[segIndex + 1] || pA;
        var lerp = function (a, b, t) { return a + (b - a) * t; };

        var curFreq = Math.round(lerp(pA.fVal, pB.fVal, localT));
        var curNote = localT >= 0.5 ? pB.note : pA.note;

        return {
          pressure: lerp(pA.pVal, pB.pVal, localT).toFixed(1) + ' kPa',
          freq: curFreq + ' Hz',
          freqHeader: curFreq === 0 ? '0 Hz (直流氣流)' : curFreq + ' Hz (' + curNote + ')',
          spl: Math.round(lerp(pA.splVal, pB.splVal, localT)) + ' dB'
        };
      }

      var curIdx = -1;
      function setPhase(idx) {
        if (curIdx === idx) return;
        curIdx = idx;
        var d = phases[idx];

        zoneLabel.textContent = d.zone;
        waveMode.textContent = d.mode;
        statusTag.textContent = d.status;

        if (ambientGlow) {
          ambientGlow.className = 'absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[320px] sm:w-[540px] h-[320px] sm:h-[540px] rounded-full blur-[130px] pointer-events-none transition-colors duration-700 ' + d.glowClass;
        }

        ['ba-part-breath','ba-part-mouthpiece','ba-part-bore','ba-part-valves','ba-part-bell'].forEach(function (id) {
          var el = root.querySelector('#' + id);
          if (el) el.setAttribute('opacity', id === d.part ? '1' : '0.35');
        });

        if (v1 && v2 && v3) {
          v1.setAttribute('y', d.valves[0] ? '62' : '52');
          v2.setAttribute('y', d.valves[1] ? '66' : '58');
          v3.setAttribute('y', d.valves[2] ? '62' : '52');
        }

        nodes.forEach(function (n, i) {
          var o = n.querySelector('.node-outer'), c = n.querySelector('.node-core');
          if (i <= idx) {
            n.classList.add('is-active');
            o.setAttribute('stroke', phases[i].color);
            c.setAttribute('fill', phases[i].color);
          } else {
            n.classList.remove('is-active');
            o.setAttribute('stroke', '#475569');
            c.setAttribute('fill', '#94a3b8');
          }
        });

        cards.forEach(function (c, i) {
          if (i === idx) {
            c.classList.add('is-active');
            gsap.to(c, {
              opacity: 1,
              scale: 1,
              y: 0,
              duration: 0.38,
              ease: 'power3.out',
              overwrite: 'auto'
            });
          } else {
            c.classList.remove('is-active');
            gsap.to(c, {
              opacity: 0,
              scale: 0.97,
              y: 14,
              duration: 0.3,
              ease: 'power2.out',
              overwrite: 'auto'
            });
          }
        });

        huds.forEach(function (b, i) {
          if (i === idx) {
            b.classList.add('is-active');
            b.classList.remove('text-slate-400');
          } else {
            b.classList.remove('is-active');
            b.classList.add('text-slate-400');
          }
        });
      }

      function onScrollUpdate(prog) {
        var c = Math.max(0, Math.min(1, prog));
        topBar.style.transform = 'scaleX(' + c + ')';
        topText.textContent = String(Math.round(c * 100)).padStart(3, '0') + '%';

        var draw = pathLength * c;
        activeTrack.style.strokeDashoffset = String(pathLength - draw);
        var pt = activeTrack.getPointAtLength(draw);
        playhead.setAttribute('transform', 'translate(' + pt.x.toFixed(1) + ',' + pt.y.toFixed(1) + ')');

        // 連續數值動態更新 (lerp)
        var interp = interpolateMetrics(c);
        mPressure.textContent = interp.pressure;
        mFreq.textContent = interp.freq;
        headerFreq.textContent = interp.freqHeader;
        mSpl.textContent = interp.spl;

        // 小號管內駐波與號口輻射環連續物理形變
        var pts = [];
        for (var i = 0; i <= 50; i++) {
          var t = i / 50;
          var x = 65 + 353 * t;
          var y = 90 + Math.sin(t * Math.PI * 2 * (1.5 + c * 4.5) - c * 16) * (3 + c * 20) * Math.sin(t * Math.PI);
          pts.push((i === 0 ? 'M ' : 'L ') + x.toFixed(1) + ' ' + y.toFixed(1));
        }
        wavePath.setAttribute('d', pts.join(' '));
        bellRings.setAttribute('opacity', (0.1 + c * 0.9).toFixed(2));

        var bestI = 0, minD = Infinity;
        ratios.forEach(function (r, idx) {
          var d = Math.abs(c - r);
          if (d < minD) { minD = d; bestI = idx; }
        });
        setPhase(bestI);
      }

      // 使用原生 position: sticky 搭配外層 420vh 跑道驅動，確保 100% 停留在一個視窗大小
      var headerOffset = parseFloat(getComputedStyle(root).getPropertyValue('--ba-header-offset')) ||
                         parseFloat(getComputedStyle(root).getPropertyValue('--eo-header-offset')) || 0;

      var mainScrollTrigger = ScrollTrigger.create({
        trigger: runway,
        start: 'top top+=' + headerOffset,
        end: 'bottom bottom',
        scrub: 1.0,
        onUpdate: function (self) { onScrollUpdate(self.progress); }
      });

      setPhase(0);
      onScrollUpdate(0);

      function scrollToPhase(phaseIndex) {
        var targetRatio = ratios[phaseIndex] !== undefined ? ratios[phaseIndex] : 0;
        var targetScrollY = mainScrollTrigger.start + (mainScrollTrigger.end - mainScrollTrigger.start) * targetRatio;
        gsap.to(window, {
          scrollTo: { y: targetScrollY, autoKill: false },
          duration: 0.9,
          ease: 'power3.inOut'
        });
      }

      nodes.forEach(function (n, i) {
        n.addEventListener('click', function () { scrollToPhase(i); });
        n.addEventListener('keydown', function (e) {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            scrollToPhase(i);
          }
        });
      });

      huds.forEach(function (b, i) {
        b.addEventListener('click', function () { scrollToPhase(i); });
      });

      cards.forEach(function (c, i) {
        c.addEventListener('click', function () { scrollToPhase(i); });
      });

      window.addEventListener('resize', function () {
        var idx = curIdx;
        curIdx = -1;
        setPhase(idx >= 0 ? idx : 0);
        ScrollTrigger.refresh();
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', ensureDependenciesAndInit);
    } else {
      ensureDependenciesAndInit();
    }
  })();
</script>
