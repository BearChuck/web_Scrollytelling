    function loadScript(src) {
      return new Promise(function (resolve) {
        if (src.indexOf('tailwindcss') !== -1 && window.tailwind) { resolve(); return; }
        if (src.indexOf('gsap.min.js') !== -1 && window.gsap) { resolve(); return; }
        if (src.indexOf('ScrollTrigger.min.js') !== -1 && window.ScrollTrigger) { resolve(); return; }
        if (src.indexOf('ScrollToPlugin.min.js') !== -1 && window.ScrollToPlugin) { resolve(); return; }

        var existing = document.querySelector('script[src="' + src + '"]');
        if (existing) {
          existing.addEventListener('load', function () { resolve(); });
          existing.addEventListener('error', function () { resolve(); });
          return;
        }

        var s = document.createElement('script');
        s.src = src;
        s.onload = function () { resolve(); };
        s.onerror = function () { resolve(); };
        document.head.appendChild(s);
      });
    }

        attempts++;

          callback();

          // 若超過超時上限 (如離線且無快取)，仍執行基礎靜態備援渲染
          callback();
        }
      }
      check();
    }

      function interpolateMetrics(progress) {
        var clamped = Math.max(0, Math.min(1, progress));
        var segIndex = 0;
        for (var i = 0; i < ratios.length - 1; i++) {
          if (clamped >= ratios[i] && clamped <= ratios[i + 1]) {
            segIndex = i;
            break;
          }

        var endRatio = ratios[segIndex + 1];
        var localT = endRatio > startRatio ? (clamped - startRatio) / (endRatio - startRatio) : 0;
        var pA = phases[segIndex];
        var pB = phases[segIndex + 1] || pA;
        var lerp = function (a, b, t) { return a + (b - a) * t; };

        var curFreq = Math.round(lerp(pA.fVal, pB.fVal, localT));
        var curNote = localT >= 0.5 ? pB.note : pA.note;

        return {
          pressure: lerp(pA.pVal, pB.pVal, localT).toFixed(1) + ' kPa',
          freq: curFreq + ' Hz',
          freqHeader: localT >= 0.5 ? pB.headerNote : pA.headerNote,
          spl: Math.round(lerp(pA.splVal, pB.splVal, localT)) + ' dB'

        curIdx = -1;
        setPhase(idx >= 0 ? idx : 0);
        initOrRefreshScrollTrigger();
      });

      window.addEventListener('load', function () {
        initOrRefreshScrollTrigger();
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', ensureDependenciesAndInit);
    } else {
      ensureDependenciesAndInit();
    }
  })();
</script>
